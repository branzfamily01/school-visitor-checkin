  async function syncOnce(){
    if(!state.sync.enabled||state.sync.syncing||!state.people.length)return;
    state.sync.syncing=true; setSyncStatus('syncing');
    try{
      if(!state.sync.eventToken) await prepareSyncTokens();
      let current=state.sync.queue.filter(q=>normalizeSyncCode(q.syncCode)===effectiveSyncCode());
      while(current.length){
        const batchItems=current.slice(0,200), records=await materializeBatch(batchItems);
        await syncApi({action:'push',eventToken:state.sync.eventToken,records});
        const sent=new Set(batchItems.map(x=>x.clientEventId));
        state.sync.queue=state.sync.queue.filter(x=>!sent.has(x.clientEventId)); save();
        current=state.sync.queue.filter(q=>normalizeSyncCode(q.syncCode)===effectiveSyncCode());
      }
      const pulled=await syncApi({action:'pull',eventToken:state.sync.eventToken});
      await applyRemoteEvents(pulled.events||[]); state.sync.lastSyncAt=nowIso(); setSyncStatus('online');
    }catch(err){ console.warn('sync failed',err); setSyncStatus('offline',err?.message||'通信できません'); }
    finally{ state.sync.syncing=false; updateSyncUi(); }
  }
  function stopSyncLoop(){ if(state.sync.timer){clearInterval(state.sync.timer);state.sync.timer=null;} }
  function startSyncLoop(){ stopSyncLoop(); state.sync.timer=setInterval(syncOnce,SYNC_INTERVAL_MS); syncOnce(); }
